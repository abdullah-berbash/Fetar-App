    <main>

        <section class="user-section">

            <label><strong>اختار اسمك:</strong></label>

            <select id="userName">

                <option value="">-- اختار من القائمة --</option>

            </select>

        </section>


        <section id="menu-container" class="menu-grid"></section>


        <section class="cart-section" id="cart-section" style="display:none;">

            <h2>طلبك الحالي</h2>

            <div id="cart-items"></div>

            <p style="margin-top:10px;">الإجمالي: <strong id="total-price">0</strong> ج.م</p>

            <button id="submit-order-btn" onclick="submitOrder()">تأكيد الطلب ✅</button>

        </section>

    </main>


    <script type="module">

        import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-app.js";

        import { getFirestore, collection, addDoc } from "https://www.gstatic.com/firebasejs/10.8.0/firebase-firestore.js";


        // ⚠️ هنا امسح الكود ده وحط الـ firebaseConfig بتاعك اللي نسخته من Firebase

        const firebaseConfig = {

            apiKey: "YOUR_API_KEY",

            authDomain: "YOUR_PROJECT.firebaseapp.com",

            projectId: "YOUR_PROJECT",

            storageBucket: "YOUR_PROJECT.appspot.com",

            messagingSenderId: "YOUR_ID",

            appId: "YOUR_APP_ID"

        };

        // ⚠️ متلمسش اللي تحت ده


        const app = initializeApp(firebaseConfig);

        const db = getFirestore(app);


        const menuItems = [

            { id: 1, name: "فول مدمس", price: 10 }, { id: 2, name: "طعمية", price: 5 },

            { id: 3, name: "بطاطس محمرة", price: 12 }, { id: 4, name: "جبنة بيضاء", price: 15 },

            { id: 5, name: "بيض مقلي", price: 12 }, { id: 6, name: "شاي", price: 5 }

        ];


        const users = ["Hesham", "Ra'fat", "Alaa", "Magdy", "Abdullah", "Fatma"];


        let cart = {};


        window.onload = () => {

            const select = document.getElementById('userName');

            users.forEach(u => select.innerHTML += `<option value="${u}">${u}</option>`);

            

            const container = document.getElementById('menu-container');

            menuItems.forEach(item => {

                container.innerHTML += `

                    <div class="card">

                        <h3>${item.name}</h3>

                        <div class="price">${item.price} ج.م</div>

                        <button onclick="addToCart(${item.id})">إضافة +</button>

                    </div>`;

            });

        };


        window.addToCart = (id) => {

            if(!document.getElementById('userName').value) { alert("اختار اسمك الأول!"); return; }

            const item = menuItems.find(i => i.id === id);

            if(cart[id]) cart[id].qty++; else cart[id] = {...item, qty: 1};

            

            document.getElementById('cart-section').style.display = 'block';

            const itemsDiv = document.getElementById('cart-items');

            itemsDiv.innerHTML = ''; let total = 0;

            Object.values(cart).forEach(i => {

                total += i.price * i.qty;

                itemsDiv.innerHTML += `<div class="cart-item"><span>${i.name} (x${i.qty})</span><span>${i.price*i.qty} ج.م</span></div>`;

            });

            document.getElementById('total-price').innerText = total;

        };


        window.submitOrder = async () => {

            if(Object.keys(cart).length === 0) return;

            const order = {

                user: document.getElementById('userName').value,

                items: Object.values(cart),

                total: Object.values(cart).reduce((s, i) => s + (i.price * i.qty), 0),

                date: new Date().toLocaleString('ar-EG')

            };

            

            try {

                await addDoc(collection(db, "orders"), order);

                alert("تم إرسال طلبك بنجاح! 🎉");

                location.reload();

            } catch (e) {

                alert("حصلت مشكلة، اتأكد إنك ظبطت الـ FirebaseConfig وإن الـ Firestore شغال.");

            }

        }

    </script>
